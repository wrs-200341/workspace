import { prisma } from "@/lib/prisma";
import {
  findVisibleWorkspaceAccount,
  requireWorkspaceViewer,
  workspaceError,
} from "@/lib/workspace/api";
import { publicVideoTask, refreshWorkspaceVideoTask } from "@/lib/workspace/production/video-tasks.server";
import { queueWorkspaceVideoTaskProcessing } from "@/lib/workspace/production/video-task-processor.server";
import { shanghaiDate, shanghaiDayRange } from "@/lib/workspace/production/video-inventory";

export const runtime = "nodejs";
export const maxDuration = 600;

export async function GET(req: Request, { params }: { params: Promise<{ id: string }> }) {
  const me = await requireWorkspaceViewer();
  if (me instanceof Response) return me;
  const { id } = await params;
  const account = await findVisibleWorkspaceAccount(id);
  if (!account) return workspaceError(404, "璐﹀彿涓嶅瓨鍦?);
  const date = new URL(req.url).searchParams.get("date")?.trim() || shanghaiDate();
  const dayRange = shanghaiDayRange(date);
  if (!dayRange) return workspaceError(400, "浠诲姟鏃ユ湡鏃犳晥");
  const tasks = await prisma.workspaceVideoTask.findMany({
    where: {
      ownerId: account.ownerId,
      account: { is: { ownerId: account.ownerId } },
      createdAt: dayRange,
    },
    include: { account: { select: { name: true } } },
    orderBy: { createdAt: "desc" },
  });
  tasks
    .filter((task) => ["queued", "prompting", "submitting"].includes(task.status))
    .forEach((task) => queueWorkspaceVideoTaskProcessing(task.id));
  tasks
    .filter((task) => ["submitted", "processing"].includes(task.status) && task.providerTaskId)
    .forEach((task) => { void refreshWorkspaceVideoTask(task); });
  return Response.json({
    success: true,
    data: {
      tasks: tasks.map((task) => ({
        ...publicVideoTask(task, { includeReferences: true }),
        accountName: task.account.name,
      })),
    },
    error: null,
  }, { headers: { "Cache-Control": "private, no-store" } });
}
import { describe, expect, it } from "vitest";
import { filterProductionQueue, preserveTaskAccountName, productionQueueGroup } from "./production-queue";

const tasks = [
  { id: "queued", status: "queued" },
  { id: "processing", status: "processing" },
  { id: "done", status: "completed" },
  { id: "error", status: "failed" },
];

describe("production queue filters", () => {
  it("classifies every non-terminal phase as active", () => {
    expect(productionQueueGroup("prompting")).toBe("active");
    expect(productionQueueGroup("submitted")).toBe("active");
  });

  it("supports an all tab and returns a new array", () => {
    const result = filterProductionQueue(tasks, "all");
    expect(result.map((task) => task.id)).toEqual(["queued", "processing", "done", "error"]);
    expect(result).not.toBe(tasks);
  });

  it("filters terminal and active groups", () => {
    expect(filterProductionQueue(tasks, "active").map((task) => task.id)).toEqual(["queued", "processing"]);
    expect(filterProductionQueue(tasks, "completed").map((task) => task.id)).toEqual(["done"]);
    expect(filterProductionQueue(tasks, "failed").map((task) => task.id)).toEqual(["error"]);
  });

  it("preserves a source account label when an action response omits it", () => {
    const responseTask = { id: "task-1", accountName: "" };
    expect(preserveTaskAccountName(responseTask, "鏉ユ簮璐﹀彿")).toEqual({
      id: "task-1",
      accountName: "鏉ユ簮璐﹀彿",
    });
    expect(responseTask.accountName).toBe("");
  });
});
