export type ProductionQueueTab = "all" | "active" | "completed" | "failed";

export interface ProductionQueueTask {
  status: string;
}

export interface UnstoredVideoQueueTask extends ProductionQueueTask {
  id: string;
  inventorySaved: boolean;
}

export function productionQueueGroup(status: string): Exclude<ProductionQueueTab, "all"> {
  if (status === "completed") return "completed";
  if (status === "failed") return "failed";
  return "active";
}

export function filterProductionQueue<T extends ProductionQueueTask>(
  tasks: readonly T[],
  tab: ProductionQueueTab,
): T[] {
  if (tab === "all") return [...tasks];
  return tasks.filter((task) => productionQueueGroup(task.status) === tab);
}

export function findFirstUnstoredVideoTask<T extends UnstoredVideoQueueTask>(
  tasks: readonly T[],
): T | null {
  return tasks.find((task) => task.status === "completed" && !task.inventorySaved) ?? null;
}

export function preserveTaskAccountName<T extends { accountName?: string }>(
  task: T,
  fallbackAccountName: string,
): T & { accountName: string } {
  return {
    ...task,
    accountName: task.accountName?.trim() || fallbackAccountName,
  };
}
